############################################
# Freraut - Eventos de regimen tributario (orden-62)
# Escalones discretos 10-80% elegidos por el senor via DA freraut_da_set_tribute.
# El evento .1 elige banda (baja 10-40 / alta 50-80) y dispara .2 o .3
# (tope vanilla: maximo 7 opciones por evento, verificado
# tsarpoints_events.12 :1302). El sujeto llega como scope:freraut_tribute_subject
# guardado en accept_effect de la DA (patron vanilla 45_seek_royal_marriage.txt:101-115).
############################################
namespace = freraut_tribute

# Evento 1: seleccion de banda de tributo
freraut_tribute.1 = {
	type = country_event
	placement = ROOT
	title = freraut_tribute.1.t
	desc = freraut_tribute.1.d
	theme = diplomacy

	immediate = {
		scope:freraut_tribute_subject = {
			save_scope_as = freraut_tribute_subject
		}
	}

	option = {
		name = freraut_tribute.1.a
		trigger_event = { id = freraut_tribute.2 popup = yes }
	}

	option = {
		name = freraut_tribute.1.b
		trigger_event = { id = freraut_tribute.3 popup = yes }
	}
}

# Evento 2: banda baja (10/20/30/40)
freraut_tribute.2 = {
	type = country_event
	placement = ROOT
	title = freraut_tribute.2.t
	desc = freraut_tribute.2.d
	theme = diplomacy

	option = {
		name = freraut_tribute.2.a
		scope:freraut_tribute_subject = {
			freraut_remove_all_tribute_pacts = yes
			create_diplomatic_pact = {
				country = root
				type = freraut_da_tribute_p10
			}
		}
	}

	option = {
		name = freraut_tribute.2.b
		scope:freraut_tribute_subject = {
			freraut_remove_all_tribute_pacts = yes
			create_diplomatic_pact = {

				type = freraut_da_tribute
			}
		}
	}

	option = {
		name = freraut_tribute.2.c
		scope:freraut_tribute_subject = {
			freraut_remove_all_tribute_pacts = yes
			create_diplomatic_pact = {
				country = root
				type = freraut_da_tribute_p30
			}
		}
	}

	option = {
		name = freraut_tribute.2.d
		scope:freraut_tribute_subject = {
			freraut_remove_all_tribute_pacts = yes
			create_diplomatic_pact = {
				country = root
				type = freraut_da_tribute_p40
			}
		}
	}
}
