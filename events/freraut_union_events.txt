############################################
# Freraut - Aporte de la union confederal (orden-64)
# La DA freraut_da_union_tribute fija UN escalon de tributo (10-80%) que se
# aplica a TODOS los estados confederados del senor, no solo al objetivo de
# la DA (pedido de Lucas). Dentro de every_subject_or_below: this = confederado
# iterado, root = el senor; pacto con first = confederado (paga), second =
# senor (recibe) — leccion L238. El evento .1 elige banda (baja 10-40 / alta
# 50-80) y dispara .2 o .3; la opcion d exime a toda la union (0%). Se
# reutilizan los pactos freraut_da_tribute* tal como quedaron en 78c1a56
# (par ya corregido en orden-63): sin tipos de pacto nuevos (A428).
# El sujeto llega como scope:freraut_union_subject guardado en accept_effect
# de la DA (patron de freraut_tribute.1, events/freraut_tribute_events.txt).
############################################
namespace = freraut_union

# Evento 1: seleccion de banda de aporte de la union
freraut_union.1 = {
	type = country_event
	placement = ROOT
	title = freraut_union.1.t
	desc = freraut_union.1.desc
	theme = diplomacy

	immediate = {
		scope:freraut_union_subject = {
			save_scope_as = freraut_union_subject
		}
	}

	option = {
		name = freraut_union.1.a
		default_option = yes
	}

	option = {
		name = freraut_union.1.b
		trigger_event = { id = freraut_union.2 popup = yes }
	}

	option = {
		name = freraut_union.1.c
		trigger_event = { id = freraut_union.3 popup = yes }
	}

	option = {
		name = freraut_union.1.d
		every_subject_or_below = {
			limit = {
				is_subject_type = subject_type_confederated_state
			}
			freraut_remove_all_tribute_pacts = yes
			set_variable = {
				name = freraut_union_exempt
				value = 1
			}
		}
	}
}
