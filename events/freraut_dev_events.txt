# Freraut - Eventos de inversion en desarrollo (orden-63)
# Escalones discretos 10-80% elegidos por el senor via DA freraut_da_dev_invest.
# El senor paga al vasallo un % de SU ingreso (direccion inversa al tributo):
# first = vasallo, second = senor, par (no, yes) en los pactos.
# El evento .1 elige banda (baja 10-40 / alta 50-80) y dispara .2 o .3;
# la opcion c suspende la inversion (0%) removiendo solo sus pactos de
# desarrollo: coexiste con el tributo (fin de la exclusion mutua, orden-64).
# El sujeto llega como scope:freraut_dev_subject guardado en accept_effect
# de la DA (patron de freraut_tribute.1, events/freraut_tribute_events.txt).
############################################
namespace = freraut_dev

# Evento 1: seleccion de banda de inversion
freraut_dev.1 = {
	type = country_event
	placement = ROOT
	title = freraut_dev.1.t
	desc = freraut_dev.1.d
	theme = diplomacy

	immediate = {
		scope:freraut_dev_subject = {
			save_scope_as = freraut_dev_subject
		}
	}

	option = {
		name = freraut_dev.1.a
		trigger_event = { id = freraut_dev.2 popup = yes }
	}

	option = {
		name = freraut_dev.1.b
		trigger_event = { id = freraut_dev.3 popup = yes }
	}

	option = {
		name = freraut_dev.1.c
		scope:freraut_dev_subject = {
			freraut_remove_all_dev_pacts = yes
		}
	}
}
