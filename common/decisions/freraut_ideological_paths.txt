# Socialista
freraut_path_socialist = {
	is_shown = {
		is_player = yes
		NOT = { has_variable = freraut_ideology_chosen }
	}
	possible = {
		is_player = yes
		NOT = { has_variable = freraut_ideology_chosen }
	}
	when_taken = {
		set_variable = freraut_ideology_chosen
		trigger_event = { id = freraut_path.1 }
	}
	ai_chance = { value = 0 }
}
