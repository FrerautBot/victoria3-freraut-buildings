############################################
# Decisiones IA de administraciones automaticas (orden-69, task-72)
# Una decision por region estrategica terrestre (36, CORR-3): la IA crea
# una Administracion freraut agrupando los estados que posee en la region.
# - CORR-1: no existe campo `hidden` en 1.15 (0 usos vanilla); el gate es
#   is_ai = yes en possible (precedente is_ai = no en decisiones de
#   expedicion vanilla) + ai_chance base 0.
# - M803 anti-spam: cooldown global de 12 meses por pais
#   (set_variable con months + gate has_variable, patron vanilla
#   00_colonial_administration_buttons.txt:34/:91-94) y techo explicito del
#   ai_chance: base 0 + modifiers acotados (max 2+10+20+30+15+5 = 82).
# - D4 (confirmada por Lucas): explotacion si la region tiene oro, petroleo,
#   caucho, carbon o hierro (CORR-2: has_potential_resource = building_X,
#   ancla 00_companies_americas.txt:129); si no, colonia.
# - D6: herencia de la forma de gobierno del senor (cadena de 9 leyes de
#   lawgroup_governance_principles, variantes antes del padre; la rama
#   colonia NO activa law_colonial_administration inline para no pisar la
#   herencia — excepcion documentada al anti-deriva contra la DA).
# - CORR-4: higiene remove_variable en on_created (patron vanilla
#   00_colonial_administration_buttons.txt:95-97).
# - possible exige >= 1 estado cedible (region sin is_capital) para evitar
#   create_dynamic_country con capital nulo.
############################################

freraut_ai_admin_region_andes = {
	# Administracion IA en la region estrategica andes (orden-69).

	possible = {
		is_ai = yes
		is_subject = no
		is_at_war = no
		NOT = { has_variable = freraut_ai_admin_cooldown }
		any_scope_state = {
			count >= 1
			region = sr:region_andes
			NOT = { is_capital = yes }
		}
	}

	ai_chance = {
		base = 0
		modifier = {
			add = 2
			any_scope_state = {
				count >= 1
				region = sr:region_andes
			}
		}
		modifier = {
			add = 10
			any_scope_state = {
				count >= 2
				region = sr:region_andes
			}
		}
		modifier = {
			add = 20
			any_scope_state = {
				count >= 3
				region = sr:region_andes
			}
		}
		modifier = {
			add = 30
			any_scope_state = {
				count >= 4
				region = sr:region_andes
			}
		}
		modifier = {
			add = 15
			any_scope_state = {
				region = sr:region_andes
				any_scope_pop = {
					pop_acceptance < acceptance_status_4
				}
			}
		}
		modifier = {
			add = 5
			NOT = {
				capital = {
					region = sr:region_andes
				}
			}
		}
	}

	when_taken = {
		set_variable = {
			name = freraut_ai_admin_cooldown
			months = 12
		}
		save_scope_as = freraut_ai_overlord
		every_scope_state = {
			limit = {
				region = sr:region_andes
			}
			set_variable = freraut_ai_admin_group
		}
		random_scope_state = {
			limit = {
				region = sr:region_andes
				owner = scope:freraut_ai_overlord
				NOT = { is_capital = yes }
			}
			save_scope_as = freraut_ai_admin_capital
		}
		ordered_primary_culture = {
			order_by = "root.culture_percent_country(this)"
			position = 0
			save_scope_as = freraut_admin_culture_1
		}
		if = {
			limit = {
				any_primary_culture = {
					count >= 2
				}
			}
			ordered_primary_culture = {
				order_by = "root.culture_percent_country(this)"
				position = 1
				save_scope_as = freraut_admin_culture_2
			}
		}
		if = {
			limit = {
				any_primary_culture = {
					count >= 3
				}
			}
			ordered_primary_culture = {
				order_by = "root.culture_percent_country(this)"
				position = 2
				save_scope_as = freraut_admin_culture_3
			}
		}
		create_dynamic_country = {
			origin = root
			country_type = colonial
			tier = principality
			culture = scope:freraut_admin_culture_1
			capital = scope:freraut_ai_admin_capital
			cede_state_trigger = {
				has_variable = freraut_ai_admin_group
				NOT = { is_capital = yes }
			}
			on_created = {
				set_variable = freraut_admin_var
				if = {
					limit = {
						exists = scope:freraut_admin_culture_2
					}
					add_primary_culture = scope:freraut_admin_culture_2
				}
				if = {
					limit = {
						exists = scope:freraut_admin_culture_3
					}
					add_primary_culture = scope:freraut_admin_culture_3
				}
				# D6: herencia de la forma de gobierno del senor (variantes
				# especificas antes del padre; has_law/activate_law anclados en
				# 00_colonial_administration_buttons.txt:98-99).
				if = {
					limit = {
						scope:freraut_ai_overlord = {
							has_law = law_type:law_social_monarchy
						}
					}
					activate_law = law_type:law_social_monarchy
				}
				else_if = {
					limit = {
						scope:freraut_ai_overlord = {
							has_law = law_type:law_presidential_republic
						}
					}
					activate_law = law_type:law_presidential_republic
				}
				else_if = {
					limit = {
						scope:freraut_ai_overlord = {
							has_law = law_type:law_parliamentary_republic
						}
					}
					activate_law = law_type:law_parliamentary_republic
				}
				else_if = {
					limit = {
						scope:freraut_ai_overlord = {
							has_law = law_type:law_theocracy
						}
					}
					activate_law = law_type:law_theocracy
				}
				else_if = {
					limit = {
						scope:freraut_ai_overlord = {
							has_law = law_type:law_council_republic
						}
					}
					activate_law = law_type:law_council_republic
				}
				else_if = {
					limit = {
						scope:freraut_ai_overlord = {
							has_law = law_type:law_corporate_state
						}
					}
					activate_law = law_type:law_corporate_state
				}
				else_if = {
					limit = {
						scope:freraut_ai_overlord = {
							has_law = law_type:law_chiefdom
						}
					}
					activate_law = law_type:law_chiefdom
				}
				else_if = {
					limit = {
						scope:freraut_ai_overlord = {
							has_law = law_type:law_colonial_administration
						}
					}
					activate_law = law_type:law_colonial_administration
				}
				else_if = {
					limit = {
						scope:freraut_ai_overlord = {
							has_law = law_type:law_monarchy
						}
					}
					activate_law = law_type:law_monarchy
				}
				# Parte C: transformacion inline colonia/explotacion (D4).
				if = {
					limit = {
						any_scope_state = {
							OR = {
								has_potential_resource = building_gold_mine
								has_potential_resource = building_gold_field
								has_potential_resource = building_oil_rig
								has_potential_resource = building_rubber_plantation
								has_potential_resource = building_coal_mine
								has_potential_resource = building_iron_mine
							}
						}
					}
					# rama explotacion (espejo freraut_da_explotation_regime
					# :701-713; anti-deriva M797)
					activate_law = law_type:law_colonial_exploitation
					add_modifier = {
						name = freraut_admin_exploitation
					}
					freraut_remove_all_tribute_pacts = yes
					create_diplomatic_pact = {
						country = scope:freraut_ai_overlord
						type = freraut_da_tribute
					}
					add_loyalists = {
						value = 0.02
					}
				}
				else = {
					# rama colonia (espejo freraut_da_colonial_regime :634-654
					# sin la activacion de law_colonial_administration: D6)
					scope:freraut_ai_overlord = {
						every_primary_culture = {
							limit = {
								NOT = { is_primary_culture_of = root }
							}
							save_temporary_scope_as = freraut_admin_temp_culture
							root = {
								add_primary_culture = scope:freraut_admin_temp_culture
							}
						}
					}
					activate_law = law_type:law_frontier_colonization
					freraut_remove_all_tribute_pacts = yes
				}
				add_modifier = {
					name = freraut_admin_base_liberty
				}
				# CORR-4: higiene de variables (patron vanilla
				# 00_colonial_administration_buttons.txt:95-97)
				every_scope_state = {
					remove_variable = freraut_ai_admin_group
				}
			}
		}
		hidden_effect = {
			create_diplomatic_pact = {
				country = scope:freraut_ai_admin_capital.owner
				type = freraut_da_create_administration
			}
		}
	}
}

			add = 30
			any_scope_state = {
				count >= 4
				region = sr:region_balkans

			add = 15
			any_scope_state = {
				region = sr:region_balkans
				any_scope_pop = {
					pop_acceptance < acceptance_status_4
				}
			}
		}
		modifier = {
			add = 5
			NOT = {
				capital = {
					region = sr:region_balkans

		}

	when_taken = {
		set_variable = {
			name = freraut_ai_admin_cooldown
			months = 12
		}

		every_scope_state = {
			limit = {
				region = sr:region_balkans
			}

		}
		random_scope_state = {
			limit = {
				region = sr:region_balkans
				owner = scope:freraut_ai_overlord
				NOT = { is_capital = yes }
			}
			save_scope_as = freraut_ai_admin_capital
		}

			order_by = "root.culture_percent_country(this)"
			position = 0
			save_scope_as = freraut_admin_culture_1
		}

			limit = {
				any_primary_culture = {
					count >= 2
				}
			}
			ordered_primary_culture = {
				order_by = "root.culture_percent_country(this)"
				position = 1
				save_scope_as = freraut_admin_culture_2
			}
		}

				any_primary_culture = {
					count >= 3
				}
			}

				position = 2
				save_scope_as = freraut_admin_culture_3
			}

			tier = principality

			add = 20

				count >= 3
				region = sr:region_central_america

		modifier = {
			add = 30
			any_scope_state = {
				count >= 4
				region = sr:region_central_america

		modifier = {
			add = 15
			any_scope_state = {
				region = sr:region_central_america
				any_scope_pop = {
					pop_acceptance < acceptance_status_4
				}
			}
		}
		modifier = {
			add = 5
			NOT = {
				capital = {
					region = sr:region_central_america
				}

		}

	when_taken = {
		set_variable = {
			name = freraut_ai_admin_cooldown
			months = 12

				region = sr:region_central_america

		}

			order_by = "root.culture_percent_country(this)"
			position = 0
			save_scope_as = freraut_admin_culture_1
		}

			limit = {
				any_primary_culture = {
					count >= 2
				}
			}
			ordered_primary_culture = {
				order_by = "root.culture_percent_country(this)"
				position = 1
				save_scope_as = freraut_admin_culture_2
			}
		}

							has_law = law_type:law_colonial_administration
						}
					}
					activate_law = law_type:law_colonial_administration

						scope:freraut_ai_overlord = {

					remove_variable = freraut_ai_admin_group

freraut_ai_admin_region_indonesia = {
	# Administracion IA en la region estrategica indonesia (orden-69).

	possible = {

		is_subject = no

		NOT = { has_variable = freraut_ai_admin_cooldown }

			count >= 1
			region = sr:region_indonesia
			NOT = { is_capital = yes }
		}
	}

	ai_chance = {
		base = 0
		modifier = {
			add = 2
			any_scope_state = {
				count >= 1
				region = sr:region_indonesia
			}
		}
		modifier = {
			add = 10
			any_scope_state = {
				count >= 2
				region = sr:region_indonesia
			}
		}
		modifier = {
			add = 20
			any_scope_state = {
				count >= 3
				region = sr:region_indonesia
			}
		}
		modifier = {
			add = 30
			any_scope_state = {
				count >= 4
				region = sr:region_indonesia
			}
		}
		modifier = {
			add = 15
			any_scope_state = {
				region = sr:region_indonesia
				any_scope_pop = {
					pop_acceptance < acceptance_status_4
				}
			}
		}
		modifier = {
			add = 5
			NOT = {
				capital = {
					region = sr:region_indonesia
				}
			}
		}
	}

	when_taken = {
		set_variable = {
			name = freraut_ai_admin_cooldown
			months = 12
		}
		save_scope_as = freraut_ai_overlord
		every_scope_state = {
			limit = {
				region = sr:region_indonesia
			}
			set_variable = freraut_ai_admin_group
		}
		random_scope_state = {
			limit = {
				region = sr:region_indonesia
				owner = scope:freraut_ai_overlord
				NOT = { is_capital = yes }
			}
			save_scope_as = freraut_ai_admin_capital
		}
		ordered_primary_culture = {
			order_by = "root.culture_percent_country(this)"
			position = 0
			save_scope_as = freraut_admin_culture_1
		}
		if = {
			limit = {
				any_primary_culture = {
					count >= 2
				}
			}
			ordered_primary_culture = {
				order_by = "root.culture_percent_country(this)"
				position = 1
				save_scope_as = freraut_admin_culture_2
			}
		}
		if = {
			limit = {
				any_primary_culture = {
					count >= 3
				}
			}
			ordered_primary_culture = {
				order_by = "root.culture_percent_country(this)"
				position = 2
				save_scope_as = freraut_admin_culture_3
			}
		}
		create_dynamic_country = {
			origin = root
			country_type = colonial
			tier = principality
			culture = scope:freraut_admin_culture_1
			capital = scope:freraut_ai_admin_capital
			cede_state_trigger = {
				has_variable = freraut_ai_admin_group
				NOT = { is_capital = yes }
			}
			on_created = {
				set_variable = freraut_admin_var
				if = {
					limit = {
						exists = scope:freraut_admin_culture_2
					}
					add_primary_culture = scope:freraut_admin_culture_2
				}
				if = {
					limit = {
						exists = scope:freraut_admin_culture_3
					}
					add_primary_culture = scope:freraut_admin_culture_3
				}
				# D6: herencia de la forma de gobierno del senor (variantes
				# especificas antes del padre; has_law/activate_law anclados en
				# 00_colonial_administration_buttons.txt:98-99).
				if = {
					limit = {
						scope:freraut_ai_overlord = {
							has_law = law_type:law_social_monarchy
						}
					}
					activate_law = law_type:law_social_monarchy
				}
				else_if = {
					limit = {
						scope:freraut_ai_overlord = {
							has_law = law_type:law_presidential_republic
						}
					}
					activate_law = law_type:law_presidential_republic
				}
				else_if = {
					limit = {
						scope:freraut_ai_overlord = {
							has_law = law_type:law_parliamentary_republic
						}
					}
					activate_law = law_type:law_parliamentary_republic
				}
				else_if = {
					limit = {
						scope:freraut_ai_overlord = {
							has_law = law_type:law_theocracy
						}
					}
					activate_law = law_type:law_theocracy
				}
				else_if = {
					limit = {
						scope:freraut_ai_overlord = {
							has_law = law_type:law_council_republic
						}
					}
					activate_law = law_type:law_council_republic
				}
				else_if = {
					limit = {
						scope:freraut_ai_overlord = {
							has_law = law_type:law_corporate_state
						}
					}
					activate_law = law_type:law_corporate_state
				}
				else_if = {
					limit = {
						scope:freraut_ai_overlord = {
							has_law = law_type:law_chiefdom
						}
					}
					activate_law = law_type:law_chiefdom
				}
				else_if = {
					limit = {
						scope:freraut_ai_overlord = {
							has_law = law_type:law_colonial_administration
						}
					}
					activate_law = law_type:law_colonial_administration
				}
				else_if = {
					limit = {
						scope:freraut_ai_overlord = {
							has_law = law_type:law_monarchy
						}
					}
					activate_law = law_type:law_monarchy
				}
				# Parte C: transformacion inline colonia/explotacion (D4).
				if = {
					limit = {
						any_scope_state = {
							OR = {
								has_potential_resource = building_gold_mine
								has_potential_resource = building_gold_field
								has_potential_resource = building_oil_rig
								has_potential_resource = building_rubber_plantation
								has_potential_resource = building_coal_mine
								has_potential_resource = building_iron_mine
							}
						}
					}
					# rama explotacion (espejo freraut_da_explotation_regime
					# :701-713; anti-deriva M797)
					activate_law = law_type:law_colonial_exploitation
					add_modifier = {
						name = freraut_admin_exploitation
					}
					freraut_remove_all_tribute_pacts = yes
					create_diplomatic_pact = {
						country = scope:freraut_ai_overlord
						type = freraut_da_tribute
					}
					add_loyalists = {
						value = 0.02
					}
				}
				else = {
					# rama colonia (espejo freraut_da_colonial_regime :634-654
					# sin la activacion de law_colonial_administration: D6)
					scope:freraut_ai_overlord = {
						every_primary_culture = {
							limit = {
								NOT = { is_primary_culture_of = root }
							}
							save_temporary_scope_as = freraut_admin_temp_culture
							root = {
								add_primary_culture = scope:freraut_admin_temp_culture
							}
						}
					}
					activate_law = law_type:law_frontier_colonization
					freraut_remove_all_tribute_pacts = yes
				}
				add_modifier = {
					name = freraut_admin_base_liberty
				}
				# CORR-4: higiene de variables (patron vanilla
				# 00_colonial_administration_buttons.txt:95-97)
				every_scope_state = {
					remove_variable = freraut_ai_admin_group
				}
			}
		}
		hidden_effect = {
			create_diplomatic_pact = {

			limit = {
				region = sr:region_nile_basin

			save_scope_as = freraut_ai_admin_capital
		}

					count >= 2
				}
			}

				position = 1

			add = 30
			any_scope_state = {
				count >= 4
				region = sr:region_north_india

		modifier = {
			add = 15
			any_scope_state = {
				region = sr:region_north_india
				any_scope_pop = {
					pop_acceptance < acceptance_status_4
				}
			}
		}
		modifier = {
			add = 5
			NOT = {
				capital = {
					region = sr:region_north_india
				}

		}

	when_taken = {
		set_variable = {
			name = freraut_ai_admin_cooldown
			months = 12
		}
		save_scope_as = freraut_ai_overlord
		every_scope_state = {
			limit = {
				region = sr:region_north_india
			}
			set_variable = freraut_ai_admin_group
		}
		random_scope_state = {
			limit = {
				region = sr:region_north_india
				owner = scope:freraut_ai_overlord
				NOT = { is_capital = yes }
			}
			save_scope_as = freraut_ai_admin_capital
		}
		ordered_primary_culture = {
			order_by = "root.culture_percent_country(this)"
			position = 0
			save_scope_as = freraut_admin_culture_1
		}

			limit = {
				any_primary_culture = {
					count >= 2
				}
			}
			ordered_primary_culture = {
				order_by = "root.culture_percent_country(this)"
				position = 1
				save_scope_as = freraut_admin_culture_2
			}
		}
		if = {
			limit = {
				any_primary_culture = {
					count >= 3
				}
			}
			ordered_primary_culture = {
				order_by = "root.culture_percent_country(this)"
				position = 2
				save_scope_as = freraut_admin_culture_3
			}

			tier = principality

			add = 30
			any_scope_state = {
				count >= 4

			add = 15
			any_scope_state = {
				region = sr:region_oceania
				any_scope_pop = {
					pop_acceptance < acceptance_status_4
				}
			}

		modifier = {
			add = 5

				capital = {
					region = sr:region_oceania

					count >= 2
				}

				position = 1
				save_scope_as = freraut_admin_culture_2
			}

freraut_ai_admin_region_west_africa = {
	# Administracion IA en la region estrategica west_africa (orden-69).

	possible = {
		is_ai = yes
		is_subject = no
		is_at_war = no
		NOT = { has_variable = freraut_ai_admin_cooldown }
		any_scope_state = {
			count >= 1
			region = sr:region_west_africa
			NOT = { is_capital = yes }
		}
	}

	ai_chance = {
		base = 0
		modifier = {
			add = 2
			any_scope_state = {
				count >= 1
				region = sr:region_west_africa
			}
		}
		modifier = {
			add = 10
			any_scope_state = {
				count >= 2
				region = sr:region_west_africa
			}
		}
		modifier = {
			add = 20
			any_scope_state = {
				count >= 3
				region = sr:region_west_africa
			}
		}
		modifier = {
			add = 30
			any_scope_state = {
				count >= 4
				region = sr:region_west_africa
			}
		}
		modifier = {
			add = 15
			any_scope_state = {
				region = sr:region_west_africa
				any_scope_pop = {
					pop_acceptance < acceptance_status_4
				}
			}
		}
		modifier = {
			add = 5
			NOT = {
				capital = {
					region = sr:region_west_africa
				}
			}
		}
	}

	when_taken = {
		set_variable = {
			name = freraut_ai_admin_cooldown
			months = 12
		}
		save_scope_as = freraut_ai_overlord
		every_scope_state = {
			limit = {
				region = sr:region_west_africa
			}
			set_variable = freraut_ai_admin_group
		}
		random_scope_state = {
			limit = {
				region = sr:region_west_africa
				owner = scope:freraut_ai_overlord
				NOT = { is_capital = yes }
			}
			save_scope_as = freraut_ai_admin_capital
		}
		ordered_primary_culture = {
			order_by = "root.culture_percent_country(this)"
			position = 0
			save_scope_as = freraut_admin_culture_1
		}
		if = {
			limit = {
				any_primary_culture = {
					count >= 2
				}
			}
			ordered_primary_culture = {
				order_by = "root.culture_percent_country(this)"
				position = 1
				save_scope_as = freraut_admin_culture_2
			}
		}
		if = {
			limit = {
				any_primary_culture = {
					count >= 3
				}
			}
			ordered_primary_culture = {
				order_by = "root.culture_percent_country(this)"
				position = 2
				save_scope_as = freraut_admin_culture_3
			}
		}
		create_dynamic_country = {
			origin = root
			country_type = colonial
			tier = principality
			culture = scope:freraut_admin_culture_1
			capital = scope:freraut_ai_admin_capital
			cede_state_trigger = {
				has_variable = freraut_ai_admin_group
				NOT = { is_capital = yes }
			}
			on_created = {
				set_variable = freraut_admin_var
				if = {
					limit = {
						exists = scope:freraut_admin_culture_2
					}
					add_primary_culture = scope:freraut_admin_culture_2
				}
				if = {
					limit = {
						exists = scope:freraut_admin_culture_3
					}
					add_primary_culture = scope:freraut_admin_culture_3
				}
				# D6: herencia de la forma de gobierno del senor (variantes
				# especificas antes del padre; has_law/activate_law anclados en
				# 00_colonial_administration_buttons.txt:98-99).
				if = {
					limit = {
						scope:freraut_ai_overlord = {
							has_law = law_type:law_social_monarchy
						}
					}
					activate_law = law_type:law_social_monarchy
				}
				else_if = {
					limit = {
						scope:freraut_ai_overlord = {
							has_law = law_type:law_presidential_republic
						}
					}
					activate_law = law_type:law_presidential_republic
				}
				else_if = {
					limit = {
						scope:freraut_ai_overlord = {
							has_law = law_type:law_parliamentary_republic
						}
					}
					activate_law = law_type:law_parliamentary_republic
				}
				else_if = {
					limit = {
						scope:freraut_ai_overlord = {
							has_law = law_type:law_theocracy
						}
					}
					activate_law = law_type:law_theocracy
				}
				else_if = {
					limit = {
						scope:freraut_ai_overlord = {
							has_law = law_type:law_council_republic
						}
					}
					activate_law = law_type:law_council_republic
				}
				else_if = {
					limit = {
						scope:freraut_ai_overlord = {
							has_law = law_type:law_corporate_state
						}
					}
					activate_law = law_type:law_corporate_state
				}
				else_if = {
					limit = {
						scope:freraut_ai_overlord = {
							has_law = law_type:law_chiefdom
						}
					}
					activate_law = law_type:law_chiefdom
				}
				else_if = {
					limit = {
						scope:freraut_ai_overlord = {
							has_law = law_type:law_colonial_administration
						}
					}
					activate_law = law_type:law_colonial_administration
				}
				else_if = {
					limit = {
						scope:freraut_ai_overlord = {
							has_law = law_type:law_monarchy
						}
					}
					activate_law = law_type:law_monarchy
				}
				# Parte C: transformacion inline colonia/explotacion (D4).
				if = {
					limit = {
						any_scope_state = {
							OR = {
								has_potential_resource = building_gold_mine
								has_potential_resource = building_gold_field
								has_potential_resource = building_oil_rig
								has_potential_resource = building_rubber_plantation
								has_potential_resource = building_coal_mine
								has_potential_resource = building_iron_mine
							}
						}
					}
					# rama explotacion (espejo freraut_da_explotation_regime
					# :701-713; anti-deriva M797)
					activate_law = law_type:law_colonial_exploitation
					add_modifier = {
						name = freraut_admin_exploitation
					}
					freraut_remove_all_tribute_pacts = yes
					create_diplomatic_pact = {
						country = scope:freraut_ai_overlord
						type = freraut_da_tribute
					}
					add_loyalists = {
						value = 0.02
					}
				}
				else = {
					# rama colonia (espejo freraut_da_colonial_regime :634-654
					# sin la activacion de law_colonial_administration: D6)
					scope:freraut_ai_overlord = {
						every_primary_culture = {
							limit = {
								NOT = { is_primary_culture_of = root }
							}
							save_temporary_scope_as = freraut_admin_temp_culture
							root = {
								add_primary_culture = scope:freraut_admin_temp_culture
							}
						}
					}
					activate_law = law_type:law_frontier_colonization
					freraut_remove_all_tribute_pacts = yes
				}
				add_modifier = {
					name = freraut_admin_base_liberty
				}
				# CORR-4: higiene de variables (patron vanilla
				# 00_colonial_administration_buttons.txt:95-97)
				every_scope_state = {
					remove_variable = freraut_ai_admin_group
				}
			}
		}
		hidden_effect = {
			create_diplomatic_pact = {
				country = scope:freraut_ai_admin_capital.owner
				type = freraut_da_create_administration
			}
		}
	}
}
