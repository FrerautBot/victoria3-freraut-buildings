building_sovereign_palace = {
	building_group = bg_sovereign_palace

	texture = "gfx/interface/icons/building_icons/building_palace.dds"

	city_type = city
	levels_per_mesh = 5
	expandable = no

	production_method_groups = {
		pmg_sovereign_palace_operations
		pmg_sovereign_palace_flats
	}

	required_construction = 1

	# A490b: 1 palacio por pais, solo en la capital (V3 tiene UNA instancia
	# del building por estado; los niveles no duplican)
	possible = {
		is_capital = yes
		# A491: no construible en colonias/territorios no incorporados
		is_incorporated = yes
	}

	can_build_private = {
		always = no
	}

	# A490b: el menu de construccion gubernamental filtra por
	# can_build_government, NO por possible (bug in-game reportado por
	# Lucas 2026-11-15: el palacio se podia construir en cualquier estado
	# pese a possible is_capital; precedentes vanilla de gate de menu:
	# skyscraper has_modifier = skyscraper_site 07_government.txt:186,
	# universidad pol_cannot_build_unis 07_government.txt:77).
	# Unicidad: country_has_building_type_levels (precedente vanilla
	# 11_private_infrastructure.txt:275) bloquea el segundo palacio y la
	# expansion (1 palacio = 1 nivel, refuerzo de expandable = no).
	can_build_government = {
		owner = {
			is_player = yes
			NOT = {
				country_has_building_type_levels = {
					target = bt:building_sovereign_palace
					value >= 1
				}
			}
		}
		is_capital = yes
		is_incorporated = yes
	}
}
