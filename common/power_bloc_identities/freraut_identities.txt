############################################
# Freraut - Confederación
############################################

identity_confederation = {
	# Modifiers del bloque
	power_bloc_modifier = {
		# Unión aduanera (mismo mercado)
		power_bloc_customs_union_bool = yes
		# Progreso de mandato por miembro
		power_bloc_mandate_progress_per_great_power_member_add = 12
		power_bloc_mandate_progress_per_major_power_member_add = 8
		power_bloc_mandate_progress_per_minor_power_member_add = 4
		power_bloc_mandate_progress_per_unrecognized_major_power_member_add = 3
		power_bloc_mandate_progress_per_unrecognized_regional_power_member_add = 2
		# Penalidad por miembro moderada (-0.5 en vez de -1 vanilla)
		power_bloc_cohesion_per_member_add = -0.5
	}

	# Bonus al líder: influencia y autoridad
	leader_modifier = {
		country_influence_add = 500
		country_authority_add = 100
		country_max_companies_add = 3
	}

	# Bonus a miembros no-líder
	non_leader_modifier = {
		state_infrastructure_mult = 0.15
		country_company_throughput_bonus_add = 0.05
	}

	# Bonus a TODOS los miembros (incluyendo líder)
	participant_modifier = {
		state_construction_mult = 0.1
	}

	# Sin requisitos de leyes
	possible = {
	}

	# Siempre visible
	visible = {
		always = yes
	}

	# Condiciones para salir
	can_leave = {
		OR = {
			is_power_bloc_leader = yes
			power_bloc = {
				leverage_advantage = {
					target = root
					value <= 30
				}
			}
		}
	}

	icon = "gfx/interface/icons/central_identity_pillars_icons/trade_league.dds"
	background = "gfx/interface/power_blocs/backgrounds/pb_background_generic.dds"

	mandate_progress = power_bloc_mandate_progress_by_rank

	on_created = {
		relations_hit_with_power_bloc_leaders = yes
	}

	# COHESIÓN - Diseñada para ser manejable pero no regalada
	cohesion = {
		# Base: 25% del máximo (más generoso que vanilla 15%)
		add = {
			add = 0.25
			multiply = define:NPowerBlocs|COHESION_TARGET_MAX
			desc = POWER_BLOC_COHESION_BASE
		}

		# Bonus por GDP compartido del líder
		add = {
			if = {
				limit = { exists = scope:with_country }
				add = "power_bloc_share_gdp_with(scope:with_country)"
			}
			else_if = {
				limit = { exists = scope:without_country }
				add = "power_bloc_share_gdp_without(scope:without_country)"
			}
			else = {
				add = power_bloc_share_gdp
			}
			multiply = define:NPowerBlocs|COHESION_TARGET_MAX
			multiply = 0.5
			desc = POWER_BLOC_COHESION_SHARE_GDP
		}

		# Penalidad por malas relaciones con el líder
		add = {
			if = {
				limit = { exists = scope:with_country }
				add = "scope:power_bloc.power_bloc_worst_leader_relations_with(scope:with_country)"
			}
			else_if = {
				limit = { exists = scope:without_country }
				add = "scope:power_bloc.power_bloc_worst_leader_relations_without(scope:without_country)"
			}
			else = {
				add = scope:power_bloc.power_bloc_worst_leader_relations
			}
			divide = define:NDiplomacy|MIN_RELATIONS
			multiply = define:NPowerBlocs|COHESION_TARGET_MAX
			multiply = -1
			max = 0
			desc = POWER_BLOC_COHESION_WORST_LEADER_RELATIONS
		}

		# Bonus por dependencia económica
		add = {
			if = {
				limit = { exists = scope:with_country }
				add = "scope:power_bloc.power_bloc_worst_economic_dependence_with(scope:with_country)"
			}
			else_if = {
				limit = { exists = scope:without_country }
				add = "scope:power_bloc.power_bloc_worst_economic_dependence_without(scope:without_country)"
			}
			else = {
				add = scope:power_bloc.power_bloc_worst_economic_dependence
			}
			add = -3.5
			divide = define:NDiplomacy|ECONOMIC_DEPENDENCE_MAX
			multiply = define:NPowerBlocs|COHESION_TARGET_MAX
			multiply = 0.5
			desc = POWER_BLOC_COHESION_ECONOMIC_DEPENDENCE
		}

		# Bonus por prestigio del líder
		add = {
			if = {
				limit = { exists = scope:with_country }
				add = "power_bloc_share_prestige_with(scope:with_country)"
			}
			else_if = {
				limit = { exists = scope:without_country }
				add = "power_bloc_share_prestige_without(scope:without_country)"
			}
			else = {
				add = power_bloc_share_prestige
			}
			multiply = define:NPowerBlocs|COHESION_TARGET_MAX
			multiply = 0.3
			desc = POWER_BLOC_COHESION_SHARE_PRESTIGE
		}
	}

	ai_weight = {
		value = 100
	}
}
