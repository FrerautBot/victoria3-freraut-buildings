# === 1) Modificar lengua de la administracion (sobre Estado Confederado) ===
da_freraut_change_confederate_language = {
	groups= { overlord }

	requires_approval = no
	should_notify_third_parties = yes
	show_effect_in_tooltip = yes

	texture = "gfx/interface/icons/diplomatic_action_icons/change_culture.dds"

	potential = {
		scope:target_country = {
			is_subject_of = root
			is_subject_type = subject_type_confederated_state
			NOT = { is_diplomatic_play_enemy_of = root }
		}
	}

	possible = {
		NOT = {
			scope:target_country = {
				any_primary_culture = {
					OR = {
						shares_language_trait_with_any_primary_culture = ROOT
						is_primary_culture_of = ROOT
					}
				}
			}
		}
		is_revolutionary = no
		scope:target_country = {
			is_revolutionary = no
		}
	}

	accept_effect = {
		scope:target_country = {
			capital = {
				save_temporary_scope_as = confederate_capital_scope
			}
			every_primary_culture = {
				prev = {
					add_radicals = {
						value = large_radicals
						culture = prev
					}
					remove_primary_culture = prev
				}
			}
		}
		ordered_primary_culture = {
			order_by = "root.culture_percent_country(this)"
			save_scope_as = main_culture_scope
		}
		every_primary_culture = {
			save_temporary_scope_as = temp_primary_culture
			scope:target_country = {
				add_primary_culture = scope:temp_primary_culture
			}
		}
		scope:target_country = {
			ordered_scope_pop = {
				limit = {
					is_pop_type = bureaucrats
				}
				order_by = total_size
				position = 0
				check_range_bounds = no
				change_pop_culture = {
					target = scope:main_culture_scope
					value = 0.05
				}
			}
			add_modifier = {
				name = modifier_changed_administrative_language
				days = normal_modifier_time
				is_decaying = yes
			}
			capital = {
				add_modifier = {
					name = modifier_changed_administrative_language_state
					days = normal_modifier_time
					is_decaying = yes
				}
			}
		}
	}

	ai = {
		evaluation_chance = { value = 0.1 }
		propose_score = { value = 5 }
		will_propose = {
			always = no
		}
	}
}

# === 3) Establecer aporte de la union (sobre Estado Confederado, orden-64) ===
# El escalon elegido en el evento se aplica a TODOS los confederados del senor
# (pedido de Lucas). Reutiliza los pactos freraut_da_tribute* de orden-62/63:
# dentro de every_subject_or_below, first = confederado (paga), second = senor
# (recibe), par (yes, no) ya corregido. Sin tipos de pacto nuevos.
freraut_da_union_tribute = {
	groups = { overlord }

	requires_approval = no
	should_notify_third_parties = yes
	show_effect_in_tooltip = yes

	texture = "gfx/interface/icons/diplomatic_action_icons/raise_payments.dds"

	potential = {
		scope:target_country = {
			is_subject_of = root
			is_subject_type = subject_type_confederated_state
		}
	}

	possible = {
		always = yes
	}

	accept_effect = {
		scope:target_country = {
			save_scope_as = freraut_union_subject
		}
		trigger_event = {
			id = freraut_union.1
			popup = yes
		}
	}

	ai = {
		evaluation_chance = { value = 0.1 }
		propose_score = { value = 1 }
		will_propose = { always = no }
	}
}
