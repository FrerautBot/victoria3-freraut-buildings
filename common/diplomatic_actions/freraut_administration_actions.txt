	groups = {
		general

	show_in_lens = no

		always = no
	}

	pact = {
		cost = 10
		has_junior_participant = yes

		}

		}
	}

		will_break = {
			always = no

freraut_da_colonial_regime = {
	groups = { overlord }

	requires_approval = no
	should_notify_third_parties = yes
	show_effect_in_tooltip = yes
	is_hostile = no

	texture = "gfx/interface/icons/diplomatic_action_icons/da_appoint_colonial_governor.dds"

	potential = {
		scope:target_country = {
			is_subject_of = root
			OR = {
				is_subject_type = subject_type_freraut_administration
				is_subject_type = subject_type_puppet
			}
		}
	}

	possible = {
		scope:target_country = {
			is_subject_of = root
			OR = {
				is_subject_type = subject_type_freraut_administration
				is_subject_type = subject_type_puppet
			}
			NOT = { has_law = law_type:law_colonial_administration }
		}
	}

	accept_effect = {
		root = {
			every_primary_culture = {
				limit = {
					NOT = { is_primary_culture_of = scope:target_country }
				}
				save_temporary_scope_as = freraut_admin_temp_culture
				scope:target_country = {
					add_primary_culture = scope:freraut_admin_temp_culture
				}
			}
		}
		scope:target_country = {
			activate_law = law_type:law_frontier_colonization
			activate_law = law_type:law_colonial_administration
			if = {
				limit = {
					has_modifier = freraut_admin_exploitation
				}
				remove_modifier = freraut_admin_exploitation
			}
			freraut_remove_all_tribute_pacts = yes
		}
	}

	ai = {
		evaluation_chance = { value = 0.1 }
		propose_score = { value = 1 }
		will_propose = { always = no }
	}
}

freraut_da_explotation_regime = {
	groups = { overlord }

	requires_approval = no
	should_notify_third_parties = yes
	show_effect_in_tooltip = yes
	is_hostile = no

	texture = "gfx/interface/icons/diplomatic_action_icons/raise_payments.dds"

	potential = {
		scope:target_country = {
			is_subject_of = root
			OR = {
				is_subject_type = subject_type_freraut_administration
				is_subject_type = subject_type_puppet
			}
		}
	}

	possible = {
		scope:target_country = {
			is_subject_of = root
			OR = {
				is_subject_type = subject_type_freraut_administration
				is_subject_type = subject_type_puppet
			}
			NOT = { has_law = law_type:law_colonial_administration }
			NOT = { has_modifier = freraut_admin_exploitation }
			NOT = { has_modifier = freraut_admin_unified_liberty }
			NOT = { has_diplomatic_pact = { who = root type = freraut_da_tribute } }
		}
	}

	accept_effect = {
		scope:target_country = {
			activate_law = law_type:law_colonial_exploitation
			add_modifier = {
				name = freraut_admin_exploitation
			}
			freraut_remove_all_tribute_pacts = yes
			create_diplomatic_pact = {
				country = root
				type = freraut_da_tribute
			}
		}
	}

	ai = {
		evaluation_chance = { value = 0.1 }
		propose_score = { value = 1 }
		will_propose = { always = no }
	}
}

freraut_da_unify_culture = {
	groups = { overlord }

	requires_approval = no
	should_notify_third_parties = yes
	show_effect_in_tooltip = yes
	is_hostile = no

	texture = "gfx/interface/icons/diplomatic_action_icons/change_culture.dds"

	potential = {
		scope:target_country = {
			is_subject_of = root
			OR = {
				is_subject_type = subject_type_freraut_administration
				is_subject_type = subject_type_puppet
			}
		}
	}

	possible = {
		scope:target_country = {
			is_subject_of = root
			OR = {
				is_subject_type = subject_type_freraut_administration
				is_subject_type = subject_type_puppet
			}
			any_primary_culture = {
				NOT = { is_primary_culture_of = root }
			}
			NOT = { has_modifier = freraut_admin_unified_liberty }
		}
	}

	accept_effect = {
		root = {
			every_primary_culture = {
				limit = {
					NOT = { is_primary_culture_of = scope:target_country }
				}
				save_temporary_scope_as = freraut_admin_temp_culture
				scope:target_country = {
					add_primary_culture = scope:freraut_admin_temp_culture
				}
			}
		}
		scope:target_country = {
			every_primary_culture = {
				limit = {
					NOT = { is_primary_culture_of = root }
				}
				prev = {
					remove_primary_culture = prev
				}
			}
			add_modifier = {
				name = freraut_admin_unified_liberty
			}
			if = {
				limit = {
					has_modifier = freraut_admin_exploitation
				}
				remove_modifier = freraut_admin_exploitation
				activate_law = law_type:law_frontier_colonization
			}
			freraut_remove_all_tribute_pacts = yes
		}
	}

	ai = {
		evaluation_chance = { value = 0.1 }
		propose_score = { value = 1 }
		will_propose = { always = no }
	}
}
