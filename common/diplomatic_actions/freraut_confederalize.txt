############################################
# Freraut - Acción Diplomática: Confederalizar Miembro
# Solo disponible para el líder de un power bloc
# Convierte instantáneamente a un miembro del bloc en Estado Confederado
############################################

confederalize_member = {
	groups = {
		power_bloc_leader
	}
	show_in_lens = no

	texture = "gfx/interface/icons/diplomatic_action_icons/force_become_subject.dds"

	selectable = {
		is_power_bloc_leader = yes
	}

	potential = {
		scope:target_country ?= {
			exists = power_bloc
		}
		power_bloc ?= {
			scope:target_country.power_bloc ?= this
		}
	}

	possible = {
		scope:target_country = {
			is_subject = no
			is_diplomatic_play_committed_participant = no
		}
	}

	accept_effect = {
		create_diplomatic_pact = {
			country = scope:target_country
			type = confederate
		}
		hidden_effect = {
			scope:target_country = {
				add_liberty_desire = -50
			}
		}
	}

	ai = {
		evaluation_chance = {
			value = 0.1
		}
		will_propose = {
			is_power_bloc_leader = yes
		}
		propose_score = {
			value = 10
		}
	}
}
